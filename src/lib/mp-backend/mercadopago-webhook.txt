import { createClientFromRequest } from "npm:@base44/sdk";

function generateTicketCode() {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let code = "PCH-";
  for (let i = 0; i < 6; i++) {
    code += chars[Math.floor(Math.random() * chars.length)];
  }
  return code;
}

Deno.serve(async (req) => {
  const base44 = createClientFromRequest(req);

  try {
    let paymentId = null;

    if (req.method === "GET") {
      const url = new URL(req.url);
      const topic = url.searchParams.get("topic");
      const id = url.searchParams.get("id");
      if (topic === "payment" && id) paymentId = id;
    } else if (req.method === "POST") {
      const body = await req.json();
      if (body.type === "payment" || (body.action && String(body.action).startsWith("payment"))) {
        paymentId = body.data && body.data.id ? String(body.data.id) : null;
      }
    }

    if (!paymentId) {
      return Response.json({ status: "ok", message: "Not a payment notification" }, { status: 200 });
    }

    const accessToken = Deno.env.get("MERCADOPAGO_ACCESS_TOKEN");
    if (!accessToken) return Response.json({ status: "ok" }, { status: 200 });

    const paymentResponse = await fetch("https://api.mercadopago.com/v1/payments/" + paymentId, {
      headers: { Authorization: "Bearer " + accessToken },
    });

    if (!paymentResponse.ok) {
      console.error("MP payment query error:", await paymentResponse.text());
      return Response.json({ status: "ok" }, { status: 200 });
    }

    const payment = await paymentResponse.json();
    const preferenceId = payment.preference_id;
    const externalRef = payment.external_reference;
    const mpStatus = payment.status;

    let order = null;
    if (externalRef) {
      try { order = await base44.asServiceRole.entities.TicketOrder.get(externalRef); }
      catch { order = null; }
    }
    if (!order && preferenceId) {
      const orders = await base44.asServiceRole.entities.TicketOrder.filter({ preference_id: preferenceId });
      if (orders.length > 0) order = orders[0];
    }

    if (!order) return Response.json({ status: "ok", message: "Order not found" }, { status: 200 });

    let orderStatus = "pending";
    if (mpStatus === "approved") orderStatus = "approved";
    else if (mpStatus === "rejected" || mpStatus === "cancelled") orderStatus = "rejected";

    if (order.status !== orderStatus) {
      const updateData = {
        status: orderStatus,
        payment_id: String(paymentId),
        buyer_email: order.buyer_email || (payment.payer && payment.payer.email) || "",
      };

      if (orderStatus === "approved") {
        updateData.approval_date = new Date().toISOString();
      }

      await base44.asServiceRole.entities.TicketOrder.update(order.id, updateData);

      if (orderStatus === "approved") {
        const event = await base44.asServiceRole.entities.Event.get(order.event_id);
        if (event) {
          const currentSold = event.tickets_sold || 0;
          await base44.asServiceRole.entities.Event.update(order.event_id, {
            tickets_sold: currentSold + order.quantity,
          });
        }

        const tickets = [];
        for (let i = 0; i < order.quantity; i++) {
          tickets.push({
            event_id: order.event_id,
            event_title: order.event_title,
            event_date: event ? event.date : "",
            order_id: order.id,
            buyer_name: order.buyer_name || "",
            buyer_email: order.buyer_email || "",
            buyer_phone: order.buyer_phone || "",
            quantity: 1,
            ticket_type: order.ticket_type || "anticipada",
            total_amount: order.unit_price,
            status: "confirmado",
            ticket_code: generateTicketCode(),
          });
        }

        if (tickets.length > 0) {
          await base44.asServiceRole.entities.Ticket.bulkCreate(tickets);
        }
      }
    }

    return Response.json({ status: "ok" }, { status: 200 });
  } catch (error) {
    console.error("webhook error:", error);
    return Response.json({ status: "ok", error: error.message }, { status: 200 });
  }
});
